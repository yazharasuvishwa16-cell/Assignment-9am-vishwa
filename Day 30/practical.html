Day 30 – Section B: Coding Assignments

1. Callback function using setTimeout

function delayedMessage(callback) {
  setTimeout(() => {
    callback("Message after 3 seconds");
  }, 3000);
}

delayedMessage((msg) => {
  console.log(msg);
});

2. Convert callback-based function into a Promise

function delayedMessagePromise() {
  return new Promise((resolve) => {
    setTimeout(() => {
      resolve("Converted to Promise");
    }, 2000);
  });
}

3. Create a Promise that resolves after 2 seconds

const promiseExample = new Promise((resolve) => {
  setTimeout(() => {
    resolve("Promise resolved after 2 seconds");
  }, 2000);
});

promiseExample.then(console.log);

4. Handle a rejected Promise using .catch()

const rejectedPromise = new Promise((resolve, reject) => {
  reject("Something went wrong");
});

rejectedPromise
  .catch(error => console.error(error));

5. Rewrite a Promise example using async/await

async function showMessage() {
  const result = await promiseExample;
  console.log(result);
}

showMessage();

6. Use try...catch with async/await

async function handleError() {
  try {
    await rejectedPromise;
  } catch (error) {
    console.error("Caught error:", error);
  }
}

handleError();

7. Function returning a Promise with random success or failure

function randomResult() {
  return new Promise((resolve, reject) => {
    const success = Math.random() > 0.5;
    if (success) {
      resolve("Success!");
    } else {
      reject("Failure!");
    }
  });
}

randomResult()
  .then(console.log)
  .catch(console.error);

8. Chain two .then() methods in a Promise

promiseExample
  .then(result => {
    console.log(result);
    return "Second then executed";
  })
  .then(message => {
    console.log(message);
  });

9. Async function that waits for two Promises

async function waitForBoth() {
  const p1 = Promise.resolve("First Promise");
  const p2 = Promise.resolve("Second Promise");

  const result1 = await p1;
  const result2 = await p2;

  console.log(result1, result2);
}

waitForBoth();

10. Explain output of an async function (using comments)

async function example() {
  console.log("Start");
  await Promise.resolve();
  console.log("End");
}

example();

/*
Output:
Start
End

Explanation:
The await pauses execution until the promise resolves,
then continues execution in order.
*/
